/* ============================================================
   1. HTML / PHP
   FILE: header.php
   PLACE: Where the ticker should appear in the header
   ============================================================ */

<div class="hli-header">
    <div class="hli-container">

        <?php
        $ticker_taps = new WP_Query([
            'post_type'      => 'ticker-tap',
            'post_status'    => 'publish',
            'posts_per_page' => -1,
            'orderby'        => 'date',
            'order'          => 'ASC',
        ]);

        if ($ticker_taps->have_posts()) :

            while ($ticker_taps->have_posts()) :
                $ticker_taps->the_post();

                $image       = get_field('notice_image__logo');
                $text        = get_field('notice_text');
                $link        = get_field('notice_bar_link');
                $bg_color    = get_field('background_color');
                $hover_color = get_field('hover_background_color');
                $text_color  = get_field('notice_text_color');

                /*
                 * Handle ACF image field:
                 * Supports Array, Attachment ID, or URL.
                 */
                if (is_array($image)) {
                    $image_url = $image['url'];
                    $image_alt = !empty($image['alt']) ? $image['alt'] : get_the_title();
                } elseif (is_numeric($image)) {
                    $image_url = wp_get_attachment_image_url($image, 'full');
                    $image_alt = get_the_title();
                } else {
                    $image_url = $image;
                    $image_alt = get_the_title();
                }

                /*
                 * Handle ACF link field:
                 * Supports Array or URL.
                 */
                if (is_array($link)) {
                    $link_url    = $link['url'];
                    $link_target = !empty($link['target']) ? $link['target'] : '_self';
                } else {
                    $link_url    = $link;
                    $link_target = '_blank';
                }
        ?>

                <a href="<?php echo esc_url($link_url); ?>"
                   target="<?php echo esc_attr($link_target); ?>">

                    <div class="wrapper"
                         style="
                            --ticker-bg: <?php echo esc_attr($bg_color); ?>;
                            --ticker-hover-bg: <?php echo esc_attr($hover_color); ?>;
                            --ticker-text-color: <?php echo esc_attr($text_color); ?>;
                         ">

                        <?php if ($image_url) : ?>
                            <img src="<?php echo esc_url($image_url); ?>"
                                 alt="<?php echo esc_attr($image_alt); ?>">
                        <?php endif; ?>

                        <?php if ($text) : ?>
                            <p><?php echo esc_html($text); ?></p>
                        <?php endif; ?>

                    </div>

                </a>

        <?php
            endwhile;

            wp_reset_postdata();

        endif;
        ?>

    </div>
</div>

/* ============================================================
   3. JAVASCRIPT
   FILE: assets/js/ticker-tap.js

   Current settings:
   - Delay: 3 seconds
   - Animation: 600ms
   - Pause on mouse hover
   ============================================================ */

document.addEventListener("DOMContentLoaded", function () {

    const container = document.querySelector(".hli-header .hli-container");

    if (!container) return;

    const slides = container.querySelectorAll(":scope > a");

    if (slides.length <= 1) return;

    const wrappers = container.querySelectorAll(":scope > a .wrapper");

    let current = 0;
    let timer = null;
    let isPaused = false;

    const delay = 3000;     // 3 seconds
    const duration = 600;   // Slide animation speed


    // Find the tallest wrapper
    let maxHeight = 0;

    wrappers.forEach(function (wrapper) {

        maxHeight = Math.max(
            maxHeight,
            wrapper.offsetHeight
        );

    });


    // Prevent layout jumping
    container.style.height = maxHeight + "px";
    container.style.overflow = "hidden";
    container.style.position = "relative";


    // Initial slide setup
    slides.forEach(function (slide, index) {

        slide.style.position = "absolute";
        slide.style.width = "100%";
        slide.style.top = "0";
        slide.style.left = "0";

        const wrapper = slide.querySelector(".wrapper");

        if (index === 0) {

            slide.style.display = "block";
            wrapper.style.transform = "translateY(0)";

        } else {

            slide.style.display = "none";
            wrapper.style.transform = "translateY(-100%)";

        }

    });


    // Slide to next item
    function nextSlide() {

        if (isPaused) return;

        const currentSlide = slides[current];
        const currentWrapper = currentSlide.querySelector(".wrapper");


        // Move to next slide
        current = (current + 1) % slides.length;

        const nextSlide = slides[current];
        const nextWrapper = nextSlide.querySelector(".wrapper");


        // Show next slide
        nextSlide.style.display = "block";


        // Put next wrapper above
        nextWrapper.style.transition = "none";
        nextWrapper.style.transform = "translateY(-100%)";


        // Force browser repaint
        nextWrapper.offsetHeight;


        // Slide current wrapper down
        currentWrapper.style.transition =
            `transform ${duration}ms ease`;

        currentWrapper.style.transform =
            "translateY(100%)";


        // Slide next wrapper down
        nextWrapper.style.transition =
            `transform ${duration}ms ease`;

        nextWrapper.style.transform =
            "translateY(0)";


        // Hide previous slide after animation
        setTimeout(function () {

            currentSlide.style.display = "none";

            currentWrapper.style.transition = "none";
            currentWrapper.style.transform = "translateY(0)";

        }, duration);

    }


    // Start timer
    function startTimer() {

        clearInterval(timer);

        timer = setInterval(function () {

            nextSlide();

        }, delay);

    }


    // Pause when mouse enters
    container.addEventListener("mouseenter", function () {

        isPaused = true;

        clearInterval(timer);

    });


    // Resume when mouse leaves
    container.addEventListener("mouseleave", function () {

        isPaused = false;

        startTimer();

    });


    // Start slider
    startTimer();

});


/* ============================================================
   4. WORDPRESS FUNCTIONS
   FILE: functions.php

   Enqueues the external CSS and JS.

   filemtime() automatically changes the asset version
   whenever the CSS/JS file is modified, so browser cache
   does not cause problems during development.
   ============================================================ */

function hli_ticker_tap_assets() {

    $css_path = get_template_directory() . '/assets/css/ticker-tap.css';
    $js_path  = get_template_directory() . '/assets/js/ticker-tap.js';

    wp_enqueue_style(
        'hli-ticker-tap',
        get_template_directory_uri() . '/assets/css/ticker-tap.css',
        [],
        filemtime( $css_path )
    );

    wp_enqueue_script(
        'hli-ticker-tap',
        get_template_directory_uri() . '/assets/js/ticker-tap.js',
        [],
        filemtime( $js_path ),
        true
    );

}

add_action( 'wp_enqueue_scripts', 'hli_ticker_tap_assets' );
